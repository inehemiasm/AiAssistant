cmake_minimum_required(VERSION 3.22.1)
project(chevere_minicpm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Debug app experiments must still benchmark optimized inference kernels.
add_compile_options(-O3)
# Statically link the isolated backend to avoid symbol collisions with LiteRT.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SRC ${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/llama.cpp-omni)
if(NOT EXISTS ${LLAMA_SRC}/CMakeLists.txt)
    message(FATAL_ERROR "Initialize pinned runtime: git submodule update --init --depth 1")
endif()
add_subdirectory(${LLAMA_SRC} ${CMAKE_BINARY_DIR}/llama)
# The pinned iOS fork calls Apple's F_NOCACHE unconditionally in LES.
# Compile a build-local copy with that optional OS hint explicitly unsupported.
# Android retains its normal file cache; expert routing/weights remain unchanged.
file(READ ${LLAMA_SRC}/src/llama-moe-slot.cpp CHEVERE_MOE_SOURCE)
string(REPLACE "fcntl(kv.second, F_NOCACHE, on ? 1 : 0)" "chevere_nocache(kv.second, on ? 1 : 0)" CHEVERE_MOE_SOURCE "${CHEVERE_MOE_SOURCE}")
string(REPLACE "fcntl(fd, F_NOCACHE, 1)" "chevere_nocache(fd, 1)" CHEVERE_MOE_SOURCE "${CHEVERE_MOE_SOURCE}")
file(WRITE ${CMAKE_BINARY_DIR}/chevere-moe-slot.cpp "#include <cerrno>\nstatic int chevere_nocache(int, int) { errno = ENOTSUP; return -1; }\n${CHEVERE_MOE_SOURCE}")
get_target_property(CHEVERE_LLAMA_SOURCES llama SOURCES)
list(REMOVE_ITEM CHEVERE_LLAMA_SOURCES llama-moe-slot.cpp)
list(APPEND CHEVERE_LLAMA_SOURCES ${CMAKE_BINARY_DIR}/chevere-moe-slot.cpp)
# The stripped expert loader upstream only provides Metal's alloc-only type.
# Give CPU expert tensors a distinct writable allocation type, never a GGUF mmap.
file(READ ${LLAMA_SRC}/src/llama-model.cpp CHEVERE_MODEL_SOURCE)
set(CHEVERE_EXPERT_GUARD "    if (it == pimpl->alloc_only_buft.end()) {")
string(FIND "${CHEVERE_MODEL_SOURCE}" "${CHEVERE_EXPERT_GUARD}" CHEVERE_GUARD_OFFSET)
if(CHEVERE_GUARD_OFFSET LESS 0)
    message(FATAL_ERROR "Pinned expert allocation adapter no longer matches upstream")
endif()
string(REPLACE "${CHEVERE_EXPERT_GUARD}" "    if (it == pimpl->alloc_only_buft.end() && (!dev || ggml_backend_dev_type(dev) == GGML_BACKEND_DEVICE_TYPE_CPU)) {\n        return ml.create_tensor_alloc(hparams, chevere_cpu_expert_buffer(), tn.str(), ne, type);\n    }\n${CHEVERE_EXPERT_GUARD}" CHEVERE_MODEL_SOURCE "${CHEVERE_MODEL_SOURCE}")
set(CHEVERE_OWN_GUARD "    if (it == pimpl->alloc_only_buft.end() || disabled) {")
string(FIND "${CHEVERE_MODEL_SOURCE}" "${CHEVERE_OWN_GUARD}" CHEVERE_OWN_OFFSET)
if(CHEVERE_OWN_OFFSET LESS 0)
    message(FATAL_ERROR "Pinned expert own-allocation adapter no longer matches upstream")
endif()
string(REPLACE "${CHEVERE_OWN_GUARD}" "    if (it == pimpl->alloc_only_buft.end() && !disabled && (!dev || ggml_backend_dev_type(dev) == GGML_BACKEND_DEVICE_TYPE_CPU)) {\n        buft_list_t list = {{ggml_backend_dev_by_type(GGML_BACKEND_DEVICE_TYPE_CPU), chevere_cpu_expert_buffer()}};\n        return ml.create_tensor(hparams, &pimpl->cpu_buft_list, pimpl->dev_input.buft_list, pimpl->dev_output.buft_list, &list, tn, ne, flags);\n    }\n${CHEVERE_OWN_GUARD}" CHEVERE_MODEL_SOURCE "${CHEVERE_MODEL_SOURCE}")
file(WRITE ${CMAKE_BINARY_DIR}/chevere-model.cpp "#include \"android-expert-buffer.h\"\n${CHEVERE_MODEL_SOURCE}")
list(REMOVE_ITEM CHEVERE_LLAMA_SOURCES llama-model.cpp)
list(APPEND CHEVERE_LLAMA_SOURCES ${CMAKE_BINARY_DIR}/chevere-model.cpp)
target_include_directories(llama PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${LLAMA_SRC}/ggml/src)
set_property(TARGET llama PROPERTY SOURCES ${CHEVERE_LLAMA_SOURCES})
set(LLAMA_INSTALL_VERSION "0.0.0")
add_subdirectory(${LLAMA_SRC}/tools/mtmd ${CMAKE_BINARY_DIR}/mtmd)
add_library(chevere_minicpm SHARED minicpm_jni.cpp)
target_include_directories(chevere_minicpm PRIVATE ${LLAMA_SRC}/include ${LLAMA_SRC}/ggml/include ${LLAMA_SRC}/tools/mtmd)
target_link_libraries(chevere_minicpm PRIVATE llama mtmd android log)
target_link_options(chevere_minicpm PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
